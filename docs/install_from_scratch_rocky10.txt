# Postgres rocky 10 canned version is 16. We want 18.
# Following https://www.postgresql.org/download/linux/redhat/:
# Install the repository RPM:

# Install PostgreSQL:
sudo dnf install -y postgresql18-server

#  sudo systemctl start postgresql-18

# Install pgSphere if using...

# Also there are some packages needed for the devel package that are in epel and crb

sudo dnf install epel-release
sudo dnf config-manager --set-enabled crb
sudo dnf install postgresql18-devel
# Go to https://github.com/postgrespro/pgsphere/releases and get latest version
sudo mkdir /opt/pgSphere
sudo chown phirst /opt/pgSphere
cd /opt/pgSphere
wget https://github.com/postgrespro/pgsphere/archive/refs/tags/1.5.2.tar.gz
tar xvzf 1.5.2.tar.gz
cd pgSphere-1.5.2
make PG_CONFIG=/usr/pgsql-18/bin/pg_config USE_HEALPIX=0
sudo make PG_CONFIG=/usr/pgsql-18/bin/pg_config USE_HEALPIX=0 install

sudo systemctl daemon-reload

/usr/pgsql-18/bin/initdb -D /data/postgres/pgdata
  vi /data/postgres/pgdata/postgresql.conf
